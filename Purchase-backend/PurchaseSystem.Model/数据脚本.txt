CREATE TABLE Ps_Users (
    Id INT PRIMARY KEY IDENTITY(1,1),
    UserNo nvarchar(20) unique not null,
	UserName nvarchar(20) not null,
	UserType int not null, -- 用户类型：1正式员工 2第三方员工
	Status int not null, -- 员工状态 1 正常在职 -1 离职
    CreateTime datetime DEFAULT GETDATE(),
);

CREATE TABLE Ps_Products (
    Id INT PRIMARY KEY IDENTITY(1,1),
    ProductCode NVARCHAR(50) UNIQUE NOT NULL,  -- 商品编码
    ProductName NVARCHAR(100) NOT NULL,        -- 商品名称
    Stock INT NOT NULL DEFAULT 0,          -- 总库存
    Price DECIMAL(10,2) NOT NULL,              -- 价格
	ImgSrc NVARCHAR(100) NOT NULL,
    IsHot BIT NOT NULL DEFAULT 0,               -- 是否热门商品
    ShardCount INT NOT NULL DEFAULT 1,          -- 库存分片数
    Status TINYINT NOT NULL DEFAULT 1,          -- 状态：1正常 0下架
    CreateTime datetime DEFAULT GETDATE(),
	UpdateTime datetime
);

-- 订单表
CREATE TABLE Ps_Orders (
    Id INT PRIMARY KEY IDENTITY(1,1),
    OrderNo BIGINT NOT NULL UNIQUE,
    UserId INT NOT NULL,
    ProductId INT NOT NULL,
    Quantity INT NOT NULL DEFAULT 1,
    TotalAmount DECIMAL(10,2) NOT NULL,
    Status INT NOT NULL DEFAULT 0,
    CreateTime DATETIME2 DEFAULT GETDATE(),
    PayTime DATETIME2 NULL,
    TransactionId NVARCHAR(100) NULL,
    
    INDEX IX_OrderNo (OrderNo),
    INDEX IX_UserId_Status (UserId, Status)
);
